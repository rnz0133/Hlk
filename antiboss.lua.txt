--=====================================================
-- 3. CƠ CHẾ ANTI-DROP THEO SỰ KIỆN (EVENT-DRIVEN)
--=====================================================
-- Vì sao đổi cách làm:
-- Cách quét theo bán kính quanh nhân vật sẽ THẤT BẠI khi Boss đá bay
-- nhân vật ra xa vị trí item rơi (nhân vật đã rời khỏi bán kính quét).
--
-- Cách mới: theo dõi TRỰC TIẾP từng Tool mà người chơi đang sở hữu
-- (nằm trong Backpack hoặc Character). Ngay khi Parent của Tool đó
-- đổi thành Workspace (nghĩa là bị rơi ra đất), script phản ứng
-- NGAY LẬP TỨC bất kể nhân vật đang ở đâu, không cần chờ vòng quét.

local trackedTools = {} -- [Tool] = Connection (AncestryChanged)

function trackTool(tool)
	if trackedTools[tool] then return end -- đã theo dõi rồi thì bỏ qua

	local connection
	connection = tool.AncestryChanged:Connect(function(_, newParent)
		-- Tool bị rơi ra Workspace trong khi Anti-Drop đang BẬT
		if antiDropEnabled and newParent == Workspace then
			-- Chờ 1 khung hình để đảm bảo vật lý rơi đã được server
			-- xử lý xong (tránh giật/giằng qua lại với server)
			task.defer(function()
				if tool.Parent == Workspace and character and character.Parent then
					-- Gán lại Tool vào Character -> Humanoid tự Equip
					-- (tương đương EquipTool())
					tool.Parent = character
				end
			end)
		end

		-- Tool đã bị hủy (destroy) hoặc rời khỏi người chơi hẳn -> ngừng theo dõi
		if newParent == nil then
			trackedTools[tool] = nil
			connection:Disconnect()
		end
	end)

	trackedTools[tool] = connection
end

local function untrackTool(tool)
	local connection = trackedTools[tool]
	if connection then
		connection:Disconnect()
		trackedTools[tool] = nil
	end
end

--------------------------------------------------------
-- Theo dõi các Tool đã có sẵn lúc script chạy
--------------------------------------------------------
for _, tool in ipairs(backpack:GetChildren()) do
	if tool:IsA("Tool") then
		trackTool(tool)
	end
end
for _, tool in ipairs(character:GetChildren()) do
	if tool:IsA("Tool") then
		trackTool(tool)
	end
end

--------------------------------------------------------

	if child:IsA("Tool") then
		trackTool(child)
	end
end)
