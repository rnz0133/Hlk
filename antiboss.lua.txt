-- Hàm cập nhật WalkSpeed dựa theo vị trí % trên thanh trượt (0 -> 1)
local function updateSpeedFromAlpha(alpha)
	alpha = math.clamp(alpha, 0, 1)
	local speed = math.floor(MIN_SPEED + (MAX_SPEED - MIN_SPEED) * alpha)

	sliderKnob.Position = UDim2.new(alpha, 0, 0.5, 0)
	speedLabel.Text = "Tốc độ: " .. speed

	if humanoid then
		humanoid.WalkSpeed = speed
	end
end

-- Khởi tạo vị trí knob theo WalkSpeed hiện tại
updateSpeedFromAlpha((humanoid.WalkSpeed - MIN_SPEED) / (MAX_SPEED - MIN_SPEED))

-- Xử lý kéo bằng ngón tay/chuột trên toàn bộ thanh trượt
local draggingSlider = false

local function beginDrag()
	draggingSlider = true
end

local function endDrag()
	draggingSlider = false
end

-- Cho phép bấm/chạm trực tiếp vào track hoặc knob để bắt đầu kéo
sliderTrack.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		beginDrag()
	end
end)

sliderKnob.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		beginDrag()
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		endDrag()
	end
end)

-- Theo dõi chuyển động ngón tay/chuột để cập nhật slider real-time
UserInputService.InputChanged:Connect(function(input)
	if not draggingSlider then return end
	if input.UserInputType ~= Enum.UserInputType.Touch
		and input.UserInputType ~= Enum.UserInputType.MouseMovement then
		return
	end

	local trackAbsPos = sliderTrack.AbsolutePosition.X
	local trackAbsSize = sliderTrack.AbsoluteSize.X
	local inputX = input.Position.X

	local alpha = (inputX - trackAbsPos) / trackAbsSize
	updateSpeedFromAlpha(alpha)
end)

--------------------------------------------------------
-- Vòng lặp quét & tự nhặt lại Tool bị rơi ra Workspace
--------------------------------------------------------
-- Cơ chế: khi Boss va chạm khiến Tool bị "unequip" và rơi ra Workspace
-- (Tool.Parent đổi từ Character/Backpack -> Workspace), ta phát hiện
-- Tool nào đang nằm trực tiếp trong Workspace và ở gần nhân vật,
-- rồi gán lại Parent để Humanoid tự động trang bị lại (tương đương EquipTool).

local function tryRecoverDroppedTools()
	if not antiDropEnabled then return end
	if not character or not rootPart or not rootPart.Parent then return end
	if not humanoid or humanoid.Health <= 0 then return end -- không nhặt khi đã chết

	-- Chỉ quét các object cấp 1 trong Workspace để tối ưu hiệu năng
	-- (Tool bị rơi luôn được đưa thẳng vào Workspace, không nằm sâu trong cây object)
	for _, obj in ipairs(Workspace:GetChildren()) do
		if obj:IsA("Tool") then
			local handle = obj:FindFirstChild("Handle")
			local refPart = handle or obj:FindFirstChildWhichIsA("BasePart", true)

			if refPart then
				local distance = (refPart.Position - rootPart.Position).Magnitude

				if distance <= ANTI_DROP_SCAN_RADIUS then
					-- Gán Tool vào Character -> Humanoid tự động Equip
					-- (tương đương hành vi của Humanoid:EquipTool())
					obj.Parent = character
				end
			end
		end
	end
end

-- Dùng vòng lặp thời gian thực nhưng có throttle (giới hạn tần suất quét)
-- để không gây giật lag trên điện thoại
local lastScanTime = 0

RunService.Heartbeat:Connect(function(deltaTime)
	lastScanTime += deltaTime
	if lastScanTime >= ANTI_DROP_SCAN_INTERVAL then
		lastScanTime = 0
		tryRecoverDroppedTools()
	end
end)
