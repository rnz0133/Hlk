local UserInputService = game:GetService("UserInputService")
local Workspace    = game:GetService("Workspace")

local LocalPlayer = Players.LocalPlayer
local camera = Workspace.CurrentCamera

local character, humanoid, rootPart

local flying = false
local flySpeed = 50      -- tốc độ bay hiện tại
local MIN_SPEED = 20
local MAX_SPEED = 300     -- kéo được lên tới rất nhanh

-- Các object vật lý dùng để bay

--------------------------------------------------------
-- Gắn lại nhân vật mỗi khi respawn
--------------------------------------------------------
local function onCharacterAdded(newChar)
	character = newChar
	humanoid  = character:WaitForChild("Humanoid")
	rootPart  = character:WaitForChild("HumanoidRootPart")

	if flying then
		flying = false
	end
end

character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
onCharacterAdded(character)
LocalPlayer.CharacterAdded:Connect(onCharacterAdded)

-- Cập nhật flySpeed theo % vị trí trên thanh trượt (0 -> 1)
local function updateSpeedFromAlpha(alpha)
	alpha = math.clamp(alpha, 0, 1)
	flySpeed = math.floor(MIN_SPEED + (MAX_SPEED - MIN_SPEED) * alpha)

	sliderKnob.Position = UDim2.new(alpha, 0, 0.5, 0)
	speedLabel.Text = "Tốc độ bay: " .. flySpeed
end

-- Khởi tạo vị trí knob theo flySpeed mặc định
updateSpeedFromAlpha((flySpeed - MIN_SPEED) / (MAX_SPEED - MIN_SPEED))

local draggingSlider = false

local function beginDrag()
	draggingSlider = true
end
local function endDrag()
	draggingSlider = false
end

sliderTrack.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		beginDrag()
	end
end)
sliderKnob.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		beginDrag()
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		endDrag()
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not draggingSlider then return end
	if input.UserInputType ~= Enum.UserInputType.Touch
		and input.UserInputType ~= Enum.UserInputType.MouseMovement then
		return
	end

	local trackAbsPos = sliderTrack.AbsolutePosition.X
	local trackAbsSize = sliderTrack.AbsoluteSize.X
	local inputX = input.Position.X

	local alpha = (inputX - trackAbsPos) / trackAbsSize
	updateSpeedFromAlpha(alpha)
end)

--=====================================================
-- 2. CƠ CHẾ BẬT / TẮT BAY (bay theo hướng camera, kể cả lên/xuống)
--=====================================================

local function startFlying()
	if not rootPart then return end

	bodyGyro = Instance.new("BodyGyro")
	bodyGyro.MaxTorque = Vector3.new(400000, 400000, 400000)
	bodyGyro.P = 3000
	bodyGyro.CFrame = rootPart.CFrame
	bodyGyro.Parent = rootPart

	bodyVelocity = Instance.new("BodyVelocity")
	bodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
	bodyVelocity.Velocity = Vector3.new(0, 0, 0)
	bodyVelocity.Parent = rootPart

	flyConnection = RunService.Heartbeat:Connect(function()
		if not rootPart or not humanoid then return end

		local camCFrame = camera.CFrame

		local flatForward = Vector3.new(camCFrame.LookVector.X, 0, camCFrame.LookVector.Z)
		if flatForward.Magnitude > 0.001 then flatForward = flatForward.Unit end
		local flatRight = Vector3.new(camCFrame.RightVector.X, 0, camCFrame.RightVector.Z)
		if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit end

		local moveDirection = humanoid.MoveDirection
		local forwardAmount = moveDirection:Dot(flatForward)
		local rightAmount   = moveDirection:Dot(flatRight)

		-- Dùng flySpeed hiện tại (đang được slider cập nhật real-time)
		local flyDirection = (camCFrame.LookVector * forwardAmount) + (flatRight * rightAmount)
		if flyDirection.Magnitude > 1 then
			flyDirection = flyDirection.Unit
		end

		bodyVelocity.Velocity = flyDirection * flySpeed
		bodyGyro.CFrame = CFrame.new(rootPart.Position, rootPart.Position + camCFrame.LookVector)
	end)
end

local function stopFlying()
	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end
	if bodyVelocity then
		bodyVelocity:Destroy()
		bodyVelocity = nil
	end
	if bodyGyro then
		bodyGyro:Destroy()
		bodyGyro = nil
	end
end

-- Bấm nút BẬT/TẮT bay bên trong popup
toggleButton.MouseButton1Click:Connect(function()
	flying = not flying

	if flying then
		toggleButton.Text = "Bay: BẬT"
		toggleButton.BackgroundColor3 = Color3.fromRGB(40, 150, 60)
		mainButton.BackgroundColor3 = Color3.fromRGB(40, 150, 60) -- nút chính cũng đổi màu để biết đang bay
		startFlying()
	else
		toggleButton.Text = "Bay: TẮT"
		toggleButton.BackgroundColor3 = Color3.fromRGB(150, 40, 40)
		mainButton.BackgroundColor3 = Color3.fromRGB(150, 40, 40)
		stopFlying()
	end
end)
