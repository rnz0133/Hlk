-- HELPER FUNCTIONS FOR DRAGGING & RESIZING
---------------------------------------------------------
local function makeDraggable(guiElement, dragHandle)
    dragHandle = dragHandle or guiElement
    local dragging, dragInput, dragStart, startPos

    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = guiElement.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    dragHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            guiElement.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function makeResizable(frame, resizeHandle, minSize)
    minSize = minSize or Vector2.new(300, 200)
    local resizing, resizeInput, resizeStart, startSize

    resizeHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            resizing = true
            resizeStart = input.Position
            startSize = frame.Size

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    resizing = false
                end
            end)
        end
    end)

    resizeHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            resizeInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == resizeInput and resizing then
            local delta = input.Position - resizeStart
            local newX = math.max(minSize.X, startSize.X.Offset + delta.X)
            local newY = math.max(minSize.Y, startSize.Y.Offset + delta.Y)
            frame.Size = UDim2.new(startSize.X.Scale, newX, startSize.Y.Scale, newY)
        end
    end)
end

-- 4. DATABASE FETCHING & ASSET CACHING
---------------------------------------------------------
local function getAssetFromFileOrUrl(url, filename)
    filename = "PhotoGallery_Cache/" .. filename
    
    if not isfolder("PhotoGallery_Cache") then
        makefolder("PhotoGallery_Cache")
    end
    
    if isfile(filename) then
        return getcustomasset(filename)
    else
        local success, result = pcall(function()
            return game:HttpGet(url)
        end)
        if success and result then
            writefile(filename, result)
            return getcustomasset(filename)
        end
    end
    return ""
end

local function loadGallery()
    -- Anti-cache JSON fetch using ?v=tick()
    local jsonUrl = "https://raw.githubusercontent.com/rnz0133/Hlk/refs/heads/main/link-anh.json?v=" .. tick()
    
    task.spawn(function()
        local success, response = pcall(function()
            return game:HttpGet(jsonUrl)
        end)
        
        if not success or not response then
            warn("Lỗi tải JSON từ Database!")
            return
        end
        
        local parseSuccess, data = pcall(function()
            return HttpService:JSONDecode(response)
        end)
        
        if not parseSuccess or type(data) ~= "table" then
            warn("Lỗi giải mã JSON!")
            return
        end
        
        -- Support both array of strings OR array of objects with 'url'/'link' key
        local imageList = {}
        for _, item in ipairs(data) do
            if type(item) == "string" then
                table.insert(imageList, item)
            elseif type(item) == "table" and (item.url or item.link or item.image) then
                table.insert(imageList, item.url or item.link or item.image)
            end
        end
        
        -- Create Image Buttons in Grid
        for idx, imgUrl in ipairs(imageList) do
            task.spawn(function()
                local fileName = "img_" .. tostring(idx) .. "_" .. string.sub(HttpService:GenerateGUID(false), 1, 8) .. ".png"
                local assetId = getAssetFromFileOrUrl(imgUrl, fileName)
                
                if assetId ~= "" then
                    local ImageBtn = Instance.new("ImageButton")
                    ImageBtn.Name = "Img_" .. tostring(idx)
                    ImageBtn.Parent = GalleryFrame
                    ImageBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
                    ImageBtn.BorderSizePixel = 0
                    ImageBtn.ScaleType = Enum.ScaleType.Crop
                    ImageBtn.Image = assetId
                    
                    ImageBtn.MouseButton1Click:Connect(function()
                        openLightbox(assetId)
                    end)
                end
            end)
        end
    end)
end

-- Initialize Gallery Load
loadGallery()
