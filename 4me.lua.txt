-- HELPER: Cache & Download Assets
---------------------------------------------------------
local function fetchCustomAsset(url)
    if type(url) ~= "string" or url == "" then
        warn("fetchCustomAsset: URL không hợp lệ (nhận được kiểu " .. type(url) .. ")")
        return nil
    end

    if not (writefile and getcustomasset) then
        return url -- Fallback if executor does not support custom assets
    end
    
    local filename = "gallery_cache_" .. string.gsub(url, "[%c%p%s]", "") .. ".png"
    if isfile and isfile(filename) then
        return getcustomasset(filename)
    end
    
    local success, response = pcall(function()
        return game:HttpGet(url)
    end)
    
    if success and response then
        pcall(function() writefile(filename, response) end)
        return getcustomasset(filename)
    else
        return url
    end
end

                ToggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)

local ResizeBottomRight = createResizeHandle("ResizeBottomRight", UDim2.new(1, -22, 1, -22), "◢")
local ResizeBottomLeft = createResizeHandle("ResizeBottomLeft", UDim2.new(0, 0, 1, -22), "◣")
local ResizeTopRight = createResizeHandle("ResizeTopRight", UDim2.new(1, -22, 0, 0), "◥")
local ResizeTopLeft = createResizeHandle("ResizeTopLeft", UDim2.new(0, 0, 0, 0), "◤")

-- Cho phép kéo giãn khung từ bất kỳ góc nào trong 4 góc, giữ nguyên góc đối diện cố định
local function setupCornerResize(handle, cornerName)
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local resizing = true

            local absPos = MainFrame.AbsolutePosition
            local absSize = MainFrame.AbsoluteSize
            local fixedX, fixedY

            if cornerName == "BottomRight" then
                fixedX, fixedY = absPos.X, absPos.Y
            elseif cornerName == "BottomLeft" then
                fixedX, fixedY = absPos.X + absSize.X, absPos.Y
            elseif cornerName == "TopRight" then
                fixedX, fixedY = absPos.X, absPos.Y + absSize.Y
            else -- TopLeft
                fixedX, fixedY = absPos.X + absSize.X, absPos.Y + absSize.Y
            end

            local moveConn
            moveConn = UserInputService.InputChanged:Connect(function(moveInput)
                if not resizing then return end
                if moveInput.UserInputType == Enum.UserInputType.MouseMovement or moveInput.UserInputType == Enum.UserInputType.Touch then
                    local cursorX, cursorY = moveInput.Position.X, moveInput.Position.Y
                    local newWidth = math.max(MIN_WIDTH, math.abs(cursorX - fixedX))
                    local newHeight = math.max(MIN_HEIGHT, math.abs(cursorY - fixedY))

                    local newLeft, newTop
                    if cornerName == "BottomRight" then
                        newLeft, newTop = fixedX, fixedY
                    elseif cornerName == "BottomLeft" then
                        newLeft, newTop = fixedX - newWidth, fixedY
                    elseif cornerName == "TopRight" then
                        newLeft, newTop = fixedX, fixedY - newHeight
                    else -- TopLeft
                        newLeft, newTop = fixedX - newWidth, fixedY - newHeight
                    end

                    MainFrame.Size = UDim2.new(0, newWidth, 0, newHeight)
                    MainFrame.Position = UDim2.new(0, newLeft + newWidth / 2, 0, newTop + newHeight / 2)
                end
            end)

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    resizing = false
                    if moveConn then moveConn:Disconnect() end
                end
            end)
        end
    end)
end

setupCornerResize(ResizeBottomRight, "BottomRight")
setupCornerResize(ResizeBottomLeft, "BottomLeft")
setupCornerResize(ResizeTopRight, "TopRight")
setupCornerResize(ResizeTopLeft, "TopLeft")

-- FETCH JSON DATABASE & LOAD GALLERY
---------------------------------------------------------
local function loadGallery()
    -- Đọc trực tiếp từ Firebase Realtime Database (cùng project với web admin gallery-admin.html)
    local jsonUrl = "https://anh-base-default-rtdb.asia-southeast1.firebasedatabase.app/images.json?v=" .. tick()
    
    task.spawn(function()
        local success, response = pcall(function()
            return game:HttpGet(jsonUrl)
        end)
        
        if success and response then
            local jsonSuccess, data = pcall(function()
                return HttpService:JSONDecode(response)
            end)
            
            if jsonSuccess and type(data) == "table" then
                local imageList = {}

                if data.images and type(data.images) == "table" then
                    data = data.images -- Hỗ trợ trường hợp dữ liệu bọc thêm 1 lớp {"images": {...}}
                end

                if #data > 0 then
                    -- Mảng thuần: string URL hoặc {"title":"...","url":"..."}
                    imageList = data
                else
                    -- Dữ liệu kiểu Firebase: { pushId1 = {url,title,order}, pushId2 = {...}, ... }
                    for _, v in pairs(data) do
                        if type(v) == "string" then
                            table.insert(imageList, {url = v})
                        elseif type(v) == "table" then
                            table.insert(imageList, v)
                        end
                    end
                    -- Sắp xếp đúng theo thứ tự đã kéo-thả trên web admin (field "order")
                    table.sort(imageList, function(a, b)
                        local oa = (type(a) == "table" and a.order) or 0
                        local ob = (type(b) == "table" and b.order) or 0
                        return oa < ob
                    end)
                end
                
                for idx, item in ipairs(imageList) do
                    task.spawn(function()
                        -- Hỗ trợ cả 2 dạng: string URL thuần, hoặc object {"title":"...","url":"..."}
                        local imgUrl = item
                        if type(item) == "table" then
                            imgUrl = item.url or item.link or item.image
                        end

                        if type(imgUrl) ~= "string" or imgUrl == "" then
                            warn("Bỏ qua phần tử #" .. idx .. ": không tìm thấy URL hợp lệ")
                            return
                        end

                        local localAsset = fetchCustomAsset(imgUrl)
                        if not localAsset then return end
                        
                        local ImgBtn = Instance.new("ImageButton")
                        ImgBtn.Name = "ImageItem_" .. tostring(idx)
                        ImgBtn.Image = localAsset
                        ImgBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 40)
                        ImgBtn.BorderSizePixel = 0
                        ImgBtn.ScaleType = Enum.ScaleType.Crop
                        ImgBtn.Parent = GalleryPage
                        
                        ImgBtn.MouseButton1Click:Connect(function()
                            LightboxImg.Image = localAsset
                            openLightbox()
                        end)
                    end)
                end
            else
                warn("Failed to parse JSON data.")
            end
        else
            warn("Failed to fetch JSON database from GitHub.")
        end
    end)
end

-- Initialize Gallery Load
loadGallery()
