-- HELPER: Cache & Download Assets
---------------------------------------------------------
-- Asset cache/download: tên file ngắn, ổn định và không tạo request trùng nhau.
local assetCache = {}
local assetLoading = {}

local function shortHash(value)
    local hash = 5381
    for i = 1, #value do
        hash = (hash * 33 + string.byte(value, i)) % 2147483647
    end
    return tostring(hash)
end

local function fetchCustomAsset(url)
    if type(url) ~= "string" or url == "" then return nil end
    if assetCache[url] then return assetCache[url] end
    if assetLoading[url] then
        repeat task.wait() until not assetLoading[url]
        return assetCache[url]
    end

    if not (writefile and getcustomasset) then
        assetCache[url] = url
        return url
    end

    assetLoading[url] = true
    local filename = "gallery_cache_" .. shortHash(url) .. ".img"
    local asset
    if isfile and isfile(filename) then
        local ok, result = pcall(getcustomasset, filename)
        if ok then asset = result end
    else
        local ok, response = pcall(function() return game:HttpGet(url) end)
        if ok and type(response) == "string" and #response > 0 then
            local wrote = pcall(writefile, filename, response)
            if wrote then
                local assetOk, result = pcall(getcustomasset, filename)
                if assetOk then asset = result end
            end
        end
    end

    assetLoading[url] = nil
    assetCache[url] = asset or url
    return assetCache[url]
end

local ResizeBottomRight = createResizeHandle("ResizeBottomRight", UDim2.new(1, -22, 1, -22), "◢")
local ResizeBottomLeft = createResizeHandle("ResizeBottomLeft", UDim2.new(0, 0, 1, -22), "◣")
local ResizeTopRight = createResizeHandle("ResizeTopRight", UDim2.new(1, -22, 0, 0), "◥")
local ResizeTopLeft = createResizeHandle("ResizeTopLeft", UDim2.new(0, 0, 0, 0), "◤")

-- Cho phép kéo giãn khung từ bất kỳ góc nào trong 4 góc, giữ nguyên góc đối diện cố định
local function setupCornerResize(handle, cornerName)
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local resizing = true

            local absPos = MainFrame.AbsolutePosition
            local absSize = MainFrame.AbsoluteSize
            local fixedX, fixedY

            if cornerName == "BottomRight" then
                fixedX, fixedY = absPos.X, absPos.Y
            elseif cornerName == "BottomLeft" then
                fixedX, fixedY = absPos.X + absSize.X, absPos.Y
            elseif cornerName == "TopRight" then
                fixedX, fixedY = absPos.X, absPos.Y + absSize.Y
            else -- TopLeft
                fixedX, fixedY = absPos.X + absSize.X, absPos.Y + absSize.Y
            end

            local moveConn
            moveConn = UserInputService.InputChanged:Connect(function(moveInput)
                if not resizing then return end
                if moveInput.UserInputType == Enum.UserInputType.MouseMovement or moveInput.UserInputType == Enum.UserInputType.Touch then
                    local cursorX, cursorY = moveInput.Position.X, moveInput.Position.Y
                    local newWidth = math.max(MIN_WIDTH, math.abs(cursorX - fixedX))
                    local newHeight = math.max(MIN_HEIGHT, math.abs(cursorY - fixedY))

                    local newLeft, newTop
                    if cornerName == "BottomRight" then
                        newLeft, newTop = fixedX, fixedY
                    elseif cornerName == "BottomLeft" then
                        newLeft, newTop = fixedX - newWidth, fixedY
                    elseif cornerName == "TopRight" then
                        newLeft, newTop = fixedX, fixedY - newHeight
                    else -- TopLeft
                        newLeft, newTop = fixedX - newWidth, fixedY - newHeight
                    end

                    MainFrame.Size = UDim2.new(0, newWidth, 0, newHeight)
                    MainFrame.Position = UDim2.new(0, newLeft + newWidth / 2, 0, newTop + newHeight / 2)
                end
            end)

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    resizing = false
                    if moveConn then moveConn:Disconnect() end
                end
            end)
        end
    end)
end

setupCornerResize(ResizeBottomRight, "BottomRight")
setupCornerResize(ResizeBottomLeft, "BottomLeft")
setupCornerResize(ResizeTopRight, "TopRight")
setupCornerResize(ResizeTopLeft, "TopLeft")

-- FETCH JSON DATABASE & LOAD GALLERY
---------------------------------------------------------
local function loadGallery()
    local jsonUrl = "https://anh-base-default-rtdb.asia-southeast1.firebasedatabase.app/images.json?v=" .. tostring(math.floor(os.clock() * 1000))
    task.spawn(function()
        local success, response = pcall(function() return game:HttpGet(jsonUrl) end)
        if not success or not response then warn("Không tải được dữ liệu gallery."); return end

        local jsonSuccess, data = pcall(function() return HttpService:JSONDecode(response) end)
        if not jsonSuccess or type(data) ~= "table" then warn("Dữ liệu gallery không hợp lệ."); return end
        if data.images and type(data.images) == "table" then data = data.images end

        local imageList = {}
        if #data > 0 then
            imageList = data
        else
            for _, value in pairs(data) do
                if type(value) == "string" then table.insert(imageList, {url = value})
                elseif type(value) == "table" then table.insert(imageList, value) end
            end
            table.sort(imageList, function(a, b) return (a.order or 0) < (b.order or 0) end)
        end

        local cards = {}
        local queue, queued = {}, {}
        local loadingWorkers = 0
        local MAX_CONCURRENT_DOWNLOADS = 3
        local PRELOAD_ROWS = 2

        local function getUrl(item)
            if type(item) == "string" then return item end
            if type(item) == "table" then return item.url or item.link or item.image end
        end

        local function makeCard(url, index)
            local card = Instance.new("ImageButton")
            card.Name = "ImageItem_" .. tostring(index)
            card.LayoutOrder = index
            card.Size = UDim2.fromOffset(104, 104)
            card.Image = ""
            card.BackgroundColor3 = Color3.fromRGB(38, 40, 49)
            card.BorderSizePixel = 0
            card.ScaleType = Enum.ScaleType.Crop
            card.AutoButtonColor = true
            card.Parent = GalleryPage

            local corner = Instance.new("UICorner")
            corner.CornerRadius = UDim.new(0, 8)
            corner.Parent = card

            local status = Instance.new("TextLabel")
            status.Size = UDim2.fromScale(1, 1)
            status.BackgroundTransparency = 1
            status.Text = "Đang tải..."
            status.TextColor3 = Color3.fromRGB(165, 169, 185)
            status.TextSize = 11
            status.Font = Enum.Font.SourceSans
            status.Parent = card

            card.MouseButton1Click:Connect(function()
                if card.Image ~= "" then
                    LightboxImg.Image = card.Image
                    openLightbox()
                end
            end)
            return card, status
        end

        local function enqueue(index)
            if queued[index] or not cards[index] then return end
            queued[index] = true
            table.insert(queue, index)
        end

        local function work()
            while true do
                while loadingWorkers < MAX_CONCURRENT_DOWNLOADS and #queue > 0 do
                    local index = table.remove(queue, 1)
                    loadingWorkers += 1
                    task.spawn(function()
                        local card, status, url = cards[index].card, cards[index].status, cards[index].url
                        local asset = fetchCustomAsset(url)
                        if card.Parent then
                            if asset then
                                card.Image = asset
                                status.Visible = false
                            else
                                status.Text = "Không tải được"
                                status.TextColor3 = Color3.fromRGB(240, 120, 120)
                            end
                        end
                        loadingWorkers -= 1
                    end)
                end
                if #queue == 0 and loadingWorkers == 0 then break end
                task.wait(0.05)
            end
        end

        for index, item in ipairs(imageList) do
            local url = getUrl(item)
            if type(url) == "string" and url ~= "" then
                local card, status = makeCard(url, index)
                cards[index] = {card = card, status = status, url = url}
            end
        end

        local function queueVisibleCards()
            local top = GalleryPage.CanvasPosition.Y
            local bottom = top + GalleryPage.AbsoluteSize.Y
            local rowHeight = GridLayout.AbsoluteCellSize.Y + GridLayout.CellPadding.Y.Offset
            local firstRow = math.max(0, math.floor(top / math.max(rowHeight, 1)) - PRELOAD_ROWS)
            local lastRow = math.ceil(bottom / math.max(rowHeight, 1)) + PRELOAD_ROWS
            local columns = math.max(1, math.floor((GalleryPage.AbsoluteSize.X + GridLayout.CellPadding.X.Offset) / (GridLayout.AbsoluteCellSize.X + GridLayout.CellPadding.X.Offset)))
            for index = firstRow * columns + 1, math.min(#cards, (lastRow + 1) * columns) do enqueue(index) end
            task.spawn(work)
        end

        GalleryPage:GetPropertyChangedSignal("CanvasPosition"):Connect(queueVisibleCards)
        GalleryPage:GetPropertyChangedSignal("AbsoluteSize"):Connect(queueVisibleCards)
        GridLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(queueVisibleCards)
        task.wait()
        queueVisibleCards()
    end)
end
